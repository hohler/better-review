+export function GiftCardInput({ onApply }: Props) {
+  const { t } = useTranslation();
+  const [code, setCode] = useState('');
+  const [error, setError] = useState<string | null>(null);
+  const [busy, setBusy] = useState(false);
+
+  async function submit() {
+    setBusy(true);
+    setError(null);
+    try {
+      onApply(await lookupGiftCard(code));
+      setCode('');
+    } catch (e) {
+      setError(e instanceof GiftCardError ? t(`giftCard.error.${e.reason}`) : t('giftCard.error.generic'));
+    } finally {
+      setBusy(false);
+    }
+  }
+
+  return (
+    <div className="gift-card">
+      <label htmlFor="gift-card-code">{t('giftCard.label')}</label>
+      <input id="gift-card-code" value={code} onChange={(e) => setCode(e.target.value)} placeholder="XXXX-XXXX-XXXX" />
+      <button type="button" disabled={busy || !code} onClick={submit}>
+        {t('giftCard.apply')}
+      </button>
+      {error && <p role="alert">{error}</p>}
+    </div>
+  );
+}

+import { http } from '../api/http';
+import { routes } from '../api/routes';
+import { isExpired, type GiftCard } from './gift-card';
+
+export type GiftCardRejection = 'not_found' | 'expired' | 'empty';
+
+export class GiftCardError extends Error {
+  constructor(public readonly reason: GiftCardRejection) {
+    super(`Gift card rejected: ${reason}`);
+  }
+}
+
+export async function lookupGiftCard(code: string): Promise<GiftCard> {
+  const normalized = code.replace(/[\s-]+/g, '').toUpperCase();
+  const response = await http.get(`${routes.giftCards}/${encodeURIComponent(normalized)}`);
+  if (response.status === 404) throw new GiftCardError('not_found');
+
+  const card: GiftCard = await response.json();
+  if (isExpired(card)) throw new GiftCardError('expired');
+  if (card.balance <= 0) throw new GiftCardError('empty');
+  return card;
+}

+export interface GiftCard {
+  code: string;
+  /** Remaining balance in cents. */
+  balance: number;
+  currency: string;
+  expiresAt: string | null;
+}
+
+export interface GiftCardResult {
+  applied: number;
+  remaining: number;
+}
+
+export function isExpired(card: GiftCard, now = new Date()): boolean {
+  return card.expiresAt !== null && new Date(card.expiresAt) < now;
+}
+
+// A gift card never pays more than the order total, the rest stays on the card.
+export function applyGiftCard(total: number, card: GiftCard | null): GiftCardResult {
+  if (!card || isExpired(card)) return { applied: 0, remaining: card?.balance ?? 0 };
+  const applied = Math.min(total, card.balance);
+  return { applied, remaining: card.balance - applied };
+}
